fn create_token_contract<'a>(
    e: &Env,
    admin: &Address,
) -> (token::Client<'a>, token::StellarAssetClient<'a>) {
    let sac = e.register_stellar_asset_contract_v2(admin.clone());
    (
        token::Client::new(e, &sac.address()),
        token::StellarAssetClient::new(e, &sac.address()),
    )
}

/// Standard setup: admin == seller, fixed_fee=0 (no fee, all to company), no deadline, no min, no cap.
fn setup_sale<'a>(
    e: &Env,
) -> (
    TokenSaleClient<'a>,
    Address,                       // admin (== seller)
    Address,                       // buyer
    Address,                       // treasury
    token::Client<'a>,             // sell_token
    token::StellarAssetClient<'a>, // sell_token_admin
    token::Client<'a>,             // buy_token
    token::StellarAssetClient<'a>, // buy_token_admin
    Address,                       // sale_id
) {
    let admin = Address::generate(e);
    let buyer = Address::generate(e);
    let treasury = Address::generate(e);
    let company = Address::generate(e);

    let (sell_token, sell_token_admin) = create_token_contract(e, &admin);
    let (buy_token, buy_token_admin) = create_token_contract(e, &buyer);

    let sale_id = e.register(TokenSale, ());
    let sale = TokenSaleClient::new(e, &sale_id);

    sale.create(
        &admin, &admin,
        &sell_token.address, &buy_token.address, &treasury,
        &company, &0i128,  // fixed_fee=0: no fee, all to company
        &1, &1,
        &0, &0, &0, // no deadline, no min, no cap
        &0, &0, // v8: MODE_DIRECT — the v7 pass-through path
    );

    let supply: i128 = 1_000 * 10_000_000;
    sell_token_admin.mint(&admin, &supply);
    buy_token_admin.mint(&buyer, &(500 * 10_000_000i128));
    sell_token.transfer(&admin, &sale_id, &supply);
    sale.set_active(&true);

    (sale, admin, buyer, treasury, sell_token, sell_token_admin, buy_token, buy_token_admin, sale_id)
}

#[test]
fn test_trade_rounding_truncation() {
    let e = Env::default();
    e.mock_all_auths();
    let admin = Address::generate(&e);
    let buyer = Address::generate(&e);
    let treasury = Address::generate(&e);
    let (sell_token, sell_token_admin) = create_token_contract(&e, &admin);
    let (buy_token, buy_token_admin) = create_token_contract(&e, &admin);
    let sale_id = e.register(TokenSale, ());
    let sale = TokenSaleClient::new(&e, &sale_id);
    sale.create(&admin, &admin, &sell_token.address, &buy_token.address, &treasury, &Address::generate(&e), &0i128, &1, &3, &0, &0, &0, &0, &0);
    sell_token_admin.mint(&admin, &(1_000 * 10_000_000i128));
    sell_token.transfer(&admin, &sale_id, &(1_000 * 10_000_000i128));
    buy_token_admin.mint(&buyer, &(100 * 10_000_000i128));
    sale.set_active(&true);

    sale.trade(&buyer, &10i128);
    assert_eq!(sell_token.balance(&buyer), 3); // 10/3 = 3.33 → 3

    let result = sale.try_trade(&buyer, &1i128);
    assert_eq!(result, Err(Ok(SaleError::TradeTooSmall))); // 1/3 → 0

    let result = sale.try_trade(&buyer, &2i128);
    assert_eq!(result, Err(Ok(SaleError::TradeTooSmall))); // 2/3 → 0

    sale.trade(&buyer, &3i128);
    assert_eq!(sell_token.balance(&buyer), 4); // 3/3 → 1
}

                sell_token: Address::generate(e),
                buy_token: Address::generate(e),
                treasury: Address::generate(e),

                buy_price: 1,
                is_active: true,
                deadline_ledger: 0,
                min_buy_amount: 0,
                max_buy_per_buyer: 0,
            },
        );
    });
}

const U: i128 = 10_000_000; // 1 unit at 7 decimals

/// Escrow setup: admin==seller, $5 fee (exercises D10), 1:1 price, deadline
/// at ledger `deadline`, supply of 1000 tokens → Target = 1000 U.
fn setup_escrow<'a>(
    e: &Env,
    deadline: u32,
) -> (TokenSaleClient<'a>, Address, Address, Address, Address, token::Client<'a>, token::Client<'a>, token::StellarAssetClient<'a>, Address) {
    e.mock_all_auths();
    let admin = Address::generate(e);
    let treasury = Address::generate(e);
    let company = Address::generate(e);
    let (sell_token, sell_token_admin) = create_token_contract(e, &admin);
    let (buy_token, buy_token_admin) = create_token_contract(e, &admin);
    let sale_id = e.register(TokenSale, ());
    let sale = TokenSaleClient::new(e, &sale_id);

    sale.create(
        &admin, &admin, &sell_token.address, &buy_token.address, &treasury, &company,
        &(5 * U), // $5 fixed fee — cost recovery, paid at trade (D10)
        &1, &1, &deadline, &0, &0,
        &1, &0, // MODE_ESCROW
    );
    let supply: i128 = 1_000 * U;
    sell_token_admin.mint(&admin, &supply);
    sell_token.transfer(&admin, &sale_id, &supply);
    sale.set_active(&true);

    let buyer = Address::generate(e);
    buy_token_admin.mint(&buyer, &(2_000 * U));
    (sale, admin, buyer, treasury, company, sell_token, buy_token, buy_token_admin, sale_id)
}

    assert_eq!(buy_token.balance(&company), 500 * U, "company receives what was raised, not the target");
    sale.claim_tokens(&buyer);
    assert_eq!(sell_token.balance(&buyer), 500 * U);
}

#[test]
fn test_v9_partial_close_surplus_withdrawable_owed_tokens_retained() {
    let e = Env::default();
    let (sale, admin, buyer, _, _, sell_token, _, _, sale_id) = setup_escrow_floor(&e, 1_000, 400 * U);
    sale.trade(&buyer, &(500 * U));
    e.ledger().with_mut(|li| li.sequence_number = 2_000);
    sale.close();
    // 1.000 deposited, 500 owed to the buyer → 500 surplus, not one stroop more.
    assert_eq!(sale.try_withdraw(&sell_token.address, &(501 * U)), Err(Ok(SaleError::ExceedsRemaining)));
    sale.withdraw(&sell_token.address, &(500 * U));
    assert_eq!(sell_token.balance(&admin), 500 * U);
    assert_eq!(sell_token.balance(&sale_id), 500 * U, "the owed tokens never leave before claim");
    assert_eq!(sale.try_withdraw(&sell_token.address, &(1 * U)), Err(Ok(SaleError::ExceedsRemaining)));
    sale.claim_tokens(&buyer);
    assert_eq!(sell_token.balance(&buyer), 500 * U);
    assert_eq!(sell_token.balance(&sale_id), 0);
}

//  19. v9.1 — achados da auditoria EXTERNA de direção (23/08/2026):
//      comprador autorizado no SAC para entrar no escrow; upgrade travado desde a
//      ativação; close() exige lastro de tokens para todos os claims.
// ═══════════════════════════════════════════════════════


/// Escrow com piso cujo token de venda tem as flags de um security token de verdade
/// (AUTH_REQUIRED, AUTH_REVOCABLE, CLAWBACK_ENABLED — as mesmas do SAC na mainnet).
/// O comprador nasce AUTORIZADO (como depois do sanduíche do backend).
fn setup_escrow_floor_regulated<'a>(
    e: &Env,
    deadline: u32,
    min_raise: i128,
) -> (TokenSaleClient<'a>, Address, Address, Address, token::Client<'a>, token::StellarAssetClient<'a>, Address) {
    use soroban_sdk::testutils::IssuerFlags;
    e.mock_all_auths();
    let admin = Address::generate(e);
    let treasury = Address::generate(e);
    let company = Address::generate(e);
    let sell_sac = e.register_stellar_asset_contract_v2(admin.clone());
    sell_sac.issuer().set_flag(IssuerFlags::RequiredFlag);
    sell_sac.issuer().set_flag(IssuerFlags::RevocableFlag);
    sell_sac.issuer().set_flag(IssuerFlags::ClawbackEnabledFlag);
    let sell_token = token::Client::new(e, &sell_sac.address());
    let sell_admin = token::StellarAssetClient::new(e, &sell_sac.address());
    let (buy_token, buy_token_admin) = create_token_contract(e, &admin);
    let sale_id = e.register(TokenSale, ());
    let sale = TokenSaleClient::new(e, &sale_id);
    sale.create(
        &admin, &admin, &sell_sac.address(), &buy_token.address, &treasury, &company,
        &0, &1, &1, &deadline, &0, &0, &1, &min_raise,
    );
    sell_admin.set_authorized(&sale_id, &true);
    sell_admin.mint(&sale_id, &(1_000 * U));
    sale.set_active(&true);
    let buyer = Address::generate(e);
    sell_admin.set_authorized(&buyer, &true);
    buy_token_admin.mint(&buyer, &(2_000 * U));
    (sale, admin, buyer, company, sell_token, sell_admin, sale_id)
}

#[test]
fn test_v91_escrow_trade_requires_sac_authorized_buyer() {
    let e = Env::default();
    let (sale, _, buyer, _, _, sell_admin, _) = setup_escrow_floor_regulated(&e, 1_000, 400 * U);
    // Carteira que o backend nunca autorizou (sem KYC) tenta entrar direto pelo RPC.
    sell_admin.set_authorized(&buyer, &false);
    assert_eq!(sale.try_trade(&buyer, &(100 * U)), Err(Ok(SaleError::BuyerNotAuthorized)));
    assert_eq!(sale.get_raise().3, 0, "nada entrou na rodada");
    // O sanduiche do backend autoriza antes da compra: passa.
    sell_admin.set_authorized(&buyer, &true);
    sale.trade(&buyer, &(100 * U));
    assert_eq!(sale.get_raise().3, 100 * U);
}

#[test]
fn test_v91_upgrade_allowed_only_before_activation_or_after_resolution() {
    let e = Env::default();
    e.mock_all_auths();
    let admin = Address::generate(&e);
    let (sell_token, sell_token_admin) = create_token_contract(&e, &admin);
    let (buy_token, _) = create_token_contract(&e, &admin);
    let sale_id = e.register(TokenSale, ());
    let sale = TokenSaleClient::new(&e, &sale_id);
    sale.create(
        &admin, &admin, &sell_token.address, &buy_token.address,
        &Address::generate(&e), &Address::generate(&e),
        &0, &1, &1, &1_000, &0, &0, &1, &(400 * U),
    );
    let hash = BytesN::from_array(&e, &[7u8; 32]);
    // Antes da ativacao nao ha rodada: o guard deixa passar (o host falha no hash

    assert_ne!(sale.try_upgrade(&hash), Err(Ok(SaleError::WrongState)));

    assert_eq!(sale.try_upgrade(&hash), Err(Ok(SaleError::WrongState)), "ativada e vazia: travado");

#[test]
fn test_v91_close_rejected_when_sell_balance_cannot_back_every_claim() {
    let e = Env::default();
    let (sale, _, buyer, _, sell_token, sell_admin, sale_id) = setup_escrow_floor_regulated(&e, 1_000, 400 * U);
    sale.trade(&buyer, &(500 * U));
    // Clawback do emissor no meio da rodada: ficam 400 tokens para 500 devidos.

    sale.claim_tokens(&buyer);
    assert_eq!(sell_token.balance(&buyer), 500 * U);
}
